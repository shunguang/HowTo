3) pkg jetson-ffmpeg
  a) in window:
  git clone https://github.com/jocover/jetson-ffmpeg.git

  b) tar and copy to ~/pkg/jetson-ffmpeg.tar
  c) untar to ~/pkg/jetson-ffmpeg  
  d) edit ~/pkg/jetson-ffmpeg/nvmpi.pc.in
  e) build & install
    cd /home/osi/pkg/jetson-ffmpeg  
    mkdir build
    cd build
    cmake .. -DLIB_NVBUFSURFACE=/usr/lib64/nvidia/libnvbufsurface.so -DLIB_NVBUFSURFTRANSFORM=/usr/lib64/nvidia/libnvbufsurftransform.so -DLIB_NVJPEG=/usr/lib64/nvidia/libnvjpeg.so -DLIB_V4L2=/usr/lib64/libv4l2.so
    make -j8
    su
    make install

5) ffmpeg
   a) in windows:
   git clone git clone git://source.ffmpeg.org/ffmpeg.git -b release/4.2 --depth=1

   b) tar and copy to ~/pkg/ffmpeg.tar
   c) untar to ~/pkg/ffmpeg
   g) build ffmpeg
      - cp ~/pkg/build-scripts/swu_fix_crlf.sh ~/pkg/ffmpeg/.
	  - cp ~/pkg/build-scripts/swu_build_configure.sh ~/pkg/ffmpeg/.
	  - cp ~/pkg/jetson-ffmpeg/ffmpeg_patches/ffmpeg4.2_nvmpi.patch  ~/pkg/ffmpeg/.

	  - cd ~/pkg/ffmpeg
	  - ~/pkg/ffmpeg$ git apply ffmpeg4.2_nvmpi.patch
	      i)  check ~/pkg/jetson-ffmpeg/ffmpeg_patches/,
          ii) check ffmpeg version to be build
		  iii) make sure teh appiled patch matches in ffmpeg version)
		  
		  
	  - ~/pkg/ffmpeg$ ./swu_fix_crlf.sh
      - ~/pkg/ffmpeg$ ./swu_build_configure.sh
  	  - ~/pkg/ffmpeg$ make j8
	  - ~/pkg/ffmpeg$ su
	  - ~/pkg/ffmpeg$ make install
